Q6 Derived column: performance_level

Create a performance_level column using these rules: Excellent (pass percentage >= 90), Good (>= 75), Average (>= 60), and Needs Improvement (< 60). Display the subject-wise result.

Python
import pandas as pd

df = pd.read_csv("subject_summary_clean.csv")

def performance_level(pass_percentage):
    if pass_percentage >= 90:
        return "Excellent"
    elif pass_percentage >= 75:
        return "Good"
    elif pass_percentage >= 60:
        return "Average"
    else:
        return "Needs Improvement"

df["performance_level"] = df["pass_percentage"].apply(
    performance_level
)

print(df[["subject_name", "pass_percentage",
          "performance_level"]].to_string(index=False))

Q11 Pipeline: student details + attendance + marks to dashboard

Design a data pipeline that combines student details, attendance, and examination marks and delivers the integrated data to storage and a reporting dashboard. Draw and label the main components and data flow.

Diagram
  +-----------------+   +-----------------+   +-----------------+
  | students.csv    |   | attendance.csv  |   | marks.csv       |
  | student_id,     |   | student_id,     |   | student_id,     |
  | name, branch    |   | classes_held,   |   | subject, marks  |
  |                 |   | classes_attended|   |                 |
  +--------+--------+   +--------+--------+   +--------+--------+
           |                     |                     |
           +---------------------+---------------------+
                                 |   (1) EXTRACT / INGEST
                                 v       Python scripts
                    +---------------------------+
                    |  (2) VALIDATE             |
                    |  valid student_id,        |
                    |  data types, required     |
                    |  fields                   |
                    +---------------------------+
                                 |
                                 v
                    +---------------------------+
                    |  (3) CLEAN                |
                    |  remove duplicates,       |
                    |  handle missing values,   |
                    |  fix formats              |
                    +---------------------------+
                                 |
                                 v
                    +---------------------------+
                    |  (4) TRANSFORM/INTEGRATE  |
                    |  JOIN on student_id       |
                    |  attendance % =           |
                    |   attended/held * 100     |
                    |  average marks            |
                    +---------------------------+
                                 |
                                 v
                    +---------------------------+
                    |  (5) LOAD -> STORAGE      |
                    |  MySQL / PostgreSQL table |
                    |  student_performance      |
                    +---------------------------+
                                 |
                                 v
                    +---------------------------+
                    |  (6) REPORTING DASHBOARD  |
                    |  Power BI / Tableau / SQL |
                    |  attendance vs marks,     |
                    |  low-attendance list      |
                    +---------------------------+

  Orchestration: Apache Airflow runs steps 1-6 on a schedule.
  Monitoring: Python logging records each step's status and errors.
Python (integration step)
import pandas as pd

students = pd.read_csv("students.csv")
attendance = pd.read_csv("attendance.csv")
marks = pd.read_csv("marks.csv")

# Clean
for d in (students, attendance, marks):
    d.drop_duplicates(inplace=True)
students = students.dropna(subset=["student_id"])

# Transform
attendance["attendance_pct"] = (
    attendance["classes_attended"] / attendance["classes_held"] * 100
).round(2)
avg_marks = marks.groupby("student_id", as_index=False)["marks"].mean()
avg_marks = avg_marks.rename(columns={"marks": "average_marks"})

# Integrate
integrated = (
    students
    .merge(attendance[["student_id", "attendance_pct"]], on="student_id", how="left")
    .merge(avg_marks, on="student_id", how="left")
)

# Load
integrated.to_csv("student_performance.csv", index=False)
print(integrated.head())
print("Integrated data saved to student_performance.csv")

import pandas as pd

# ---------- EXTRACT ----------
df = pd.read_csv("dataset.csv")
print("Extracted data shape:", df.shape)

Q13 ETL workflow with logging

Implement an ETL workflow that extracts data, cleans and transforms it, loads the result into a destination CSV file, and prints or logs the pipeline execution status.

Python
import logging
import pandas as pd

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
    handlers=[logging.FileHandler("etl_pipeline.log"),
              logging.StreamHandler()],
    force=True
)

def extract(path):
    logging.info("EXTRACT started: reading %s", path)
    df = pd.read_csv(path)
    logging.info("EXTRACT completed: %d rows, %d columns", df.shape[0], df.shape[1])
    return df

def transform(df):
    logging.info("TRANSFORM started")
    before = len(df)
    df = df.drop_duplicates()
    logging.info("Removed %d duplicate rows", before - len(df))

    missing = int(df.isnull().sum().sum())
    for col in df.columns:
        if pd.api.types.is_numeric_dtype(df[col]):
            df[col] = df[col].fillna(df[col].mean())
        else:
            df[col] = df[col].fillna("Unknown")
    logging.info("Filled %d missing values", missing)

    # standardize column names and text values
    df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")
    for col in df.columns:
        if not pd.api.types.is_numeric_dtype(df[col]):
            df[col] = df[col].astype(str).str.strip()

    # example derived attribute (only if the column exists)
    if "pass_percentage" in df.columns:
        df["result_status"] = df["pass_percentage"].apply(
            lambda p: "Good" if p >= 75 else "Needs Attention")
        logging.info("Added derived column result_status")

    logging.info("TRANSFORM completed: %d rows", len(df))
    return df

def load(df, path):
    logging.info("LOAD started: writing %s", path)
    df.to_csv(path, index=False)
    logging.info("LOAD completed: %d rows saved", len(df))

def run_pipeline(source, destination):
    logging.info("===== ETL PIPELINE STARTED =====")
    try:
        data = extract(source)
        data = transform(data)
        load(data, destination)
        logging.info("===== ETL PIPELINE COMPLETED SUCCESSFULLY =====")
    except FileNotFoundError:
        logging.error("Source file not found: %s", source)
        logging.error("===== ETL PIPELINE FAILED =====")
    except Exception as e:
        logging.error("Pipeline error: %s", e)
        logging.error("===== ETL PIPELINE FAILED =====")

run_pipeline("dataset.csv", "transformed_dataset.csv")

# --- Explanation ---
- Sources: Academic data comes from several systems: student master data in
  MySQL, and attendance, marks and course files in CSV/Excel.
- Ingestion: Python scripts read each source on a schedule and pass the data on.
- Processing: Pandas validates IDs and types, removes duplicates, fills missing
  values, joins datasets on student_id / subject_code, and computes KPIs such as
  average pass %, total failures, absentees, performance_level and academic_risk.
- Storage: Raw files are kept unchanged for traceability, and the cleaned,
  integrated tables are loaded into MySQL/PostgreSQL.
- Analysis & visualization: SQL queries and dashboards show subject-wise
  results and the subjects that need intervention.
- Orchestration & monitoring: Airflow runs the steps in order and retries
  failures. Logging records each run. Access control protects student data.
- Review criteria: performance (fast queries), scalability (Spark/HDFS when
  data grows), reliability (logging, retries, backups) and stakeholder needs
  (dashboards for faculty and management).

# --- Python (integration step) ---
import pandas as pd

students = pd.read_csv("students.csv")
attendance = pd.read_csv("attendance.csv")
marks = pd.read_csv("marks.csv")

# Clean
for d in (students, attendance, marks):
    d.drop_duplicates(inplace=True)
students = students.dropna(subset=["student_id"])

# Transform
attendance["attendance_pct"] = (
    attendance["classes_attended"] / attendance["classes_held"] * 100
).round(2)
avg_marks = marks.groupby("student_id", as_index=False)["marks"].mean()
avg_marks = avg_marks.rename(columns={"marks": "average_marks"})

# Integrate
integrated = (
    students
    .merge(attendance[["student_id", "attendance_pct"]], on="student_id", how="left")
    .merge(avg_marks, on="student_id", how="left")
)

# Load
integrated.to_csv("student_performance.csv", index=False)
print(integrated.head())
print("Integrated data saved to student_performance.csv")

# ===================== Q13) ETL workflow with logging =====================

# --- Python ---
import logging
import pandas as pd

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
    handlers=[logging.FileHandler("etl_pipeline.log"),
              logging.StreamHandler()],
    force=True
)

def extract(path):
    logging.info("EXTRACT started: reading %s", path)
    df = pd.read_csv(path)
    logging.info("EXTRACT completed: %d rows, %d columns", df.shape[0], df.shape[1])
    return df

def transform(df):
    logging.info("TRANSFORM started")
    before = len(df)
    df = df.drop_duplicates()
    logging.info("Removed %d duplicate rows", before - len(df))

    missing = int(df.isnull().sum().sum())
    for col in df.columns:
        if pd.api.types.is_numeric_dtype(df[col]):
            df[col] = df[col].fillna(df[col].mean())
        else:
            df[col] = df[col].fillna("Unknown")
    logging.info("Filled %d missing values", missing)

    # standardize column names and text values
    df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")
    for col in df.columns:
        if not pd.api.types.is_numeric_dtype(df[col]):
            df[col] = df[col].astype(str).str.strip()

    # example derived attribute (only if the column exists)
    if "pass_percentage" in df.columns:
        df["result_status"] = df["pass_percentage"].apply(
            lambda p: "Good" if p >= 75 else "Needs Attention")
        logging.info("Added derived column result_status")

    logging.info("TRANSFORM completed: %d rows", len(df))
    return df

def load(df, path):
    logging.info("LOAD started: writing %s", path)
    df.to_csv(path, index=False)
    logging.info("LOAD completed: %d rows saved", len(df))

def run_pipeline(source, destination):
    logging.info("===== ETL PIPELINE STARTED =====")
    try:
        data = extract(source)
        data = transform(data)
        load(data, destination)
        logging.info("===== ETL PIPELINE COMPLETED SUCCESSFULLY =====")
    except FileNotFoundError:
        logging.error("Source file not found: %s", source)
        logging.error("===== ETL PIPELINE FAILED =====")
    except Exception as e:
        logging.error("Pipeline error: %s", e)
        logging.error("===== ETL PIPELINE FAILED =====")

run_pipeline("dataset.csv", "transformed_dataset.csv")
